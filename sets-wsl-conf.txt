# /etc/wsl.conf
[boot]
systemd=true

[user]
default=<username>

[interop]
enabled = true
appendWindowsPath = false

# ------------------------------------------------------------------------------
# NOTE: 
# - 'enabled = true' ensures WSL binary interoperability (binfmt_misc) is
#   active so Linux can execute Windows .exe binaries (like Code.exe).
# - 'appendWindowsPath = false' keeps Linux $PATH clean from Windows environment
#   pollution while allowing targeted additions.
#
# To make the VS Code 'code' command available inside WSL:
# Create /etc/profile.d/vscode.sh:
# export PATH="$PATH:/mnt/c/Users/<your-windows-user>/AppData/Local/Programs/Microsoft VS Code/bin"
# ------------------------------------------------------------------------------
